  n ["n"] Builtin(NewWorkspace) "new workspace"
  r ["r"] Builtin(RenameWorkspace) "rename workspace"
  x ["x"] Builtin(CloseWorkspace) "close workspace"
  s ["s"] Builtin(WorkspacePicker) "switch workspace"
  j ["j"] Builtin(NextWorkspace) "next workspace"
  k ["k"] Builtin(PreviousWorkspace) "previous workspace"
  shift+j ["shift+j"] Builtin(MoveWorkspaceNext) "move workspace down"
  shift+k ["shift+k"] Builtin(MoveWorkspacePrevious) "move workspace up"
  g ["g"] Builtin(ToggleGroup) "expand/collapse group"

  e ["e"] Builtin(EditScrollback) "edit scrollback"

prefix ? keybinds
prefix q detach
prefix 1..9 switch tab 1-9
